import re



def check_password_strength(password):
    if len(password) < 8:
        return "Weak: Password must be atleast 8 charactors Long"

    if not any(char.isdigit() for char in password):
        return "Weak:password must be contain digit"

    if not any(char.isupper() for char in password):
        return "Weak:Password must be  containan upper letter"

    if not any(char.islower() for char in password):
        return "Weak:password most be contain a lower letter"

    if not re.search(r'[!@#$%^&()<>.?]',password):
        return "Medium:password must be contain a special charactor"

    return "Strong:Your password must be strong and secure!"

def password_checker():
    print("Wellcome to the password strenght checker")

    while True:
        password = input("Enter your password  (or type 'exit' to quit) : ")

        if password.lower() == 'exit':
            print("Thank you for using this tool! GoodBye!")
            break

        result = check_password_strength(password)
        print(result)


if __name__ == "__main__":
    password_checker()
